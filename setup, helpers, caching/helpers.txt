INPUT_ARTIFACT_INDEX = {}


def build_input_artifact_index():
    """
    Scan /kaggle/input for prior output files stored under ARTIFACT_ROOT_NAME
    so experiments can be reloaded after notebook restart.
    """
    global INPUT_ARTIFACT_INDEX
    INPUT_ARTIFACT_INDEX = {}

    for dirname, _, filenames in os.walk("/kaggle/input"):
        marker = f"/{ARTIFACT_ROOT_NAME}/"
        if marker not in dirname:
            continue

        relative_dir = dirname.split(marker, 1)[1]

        for filename in filenames:
            relative_path = os.path.join(relative_dir, filename) if relative_dir else filename
            full_path = os.path.join(dirname, filename)

            if relative_path not in INPUT_ARTIFACT_INDEX:
                INPUT_ARTIFACT_INDEX[relative_path] = full_path


build_input_artifact_index()

def normalize_experiment_name(name: str) -> str:
    """
    Make experiment names filesystem-friendly but still readable.
    """
    cleaned = re.sub(r"[^A-Za-z0-9._-]+", "_", name.strip())
    return cleaned.strip("_")


def artifact_cache_key(name, n_games, base_seed, max_steps):
    return (normalize_experiment_name(name), int(n_games), int(base_seed), int(max_steps))


def artifact_rel_dir(name, n_games, base_seed, max_steps):
    safe_name = normalize_experiment_name(name)
    return os.path.join(
        "experiments",
        safe_name,
        f"n_{n_games}_seed_{base_seed}_steps_{max_steps}",
    )


def artifact_rel_paths(name, n_games, base_seed, max_steps):
    root = artifact_rel_dir(name, n_games, base_seed, max_steps)
    return {
        "summary": os.path.join(root, "summary.json"),
        "games": os.path.join(root, "games.json.gz"),
        "replays": os.path.join(root, "replays.json.gz"),
    }


def games_table_rel_path(name, n_games, base_seed, max_steps):
    root = artifact_rel_dir(name, n_games, base_seed, max_steps)
    return os.path.join(root, "games_table.csv")


def artifact_write_path(rel_path):
    return os.path.join(ARTIFACT_WRITE_ROOT, rel_path)


def artifact_read_path(rel_path):
    candidate = artifact_write_path(rel_path)
    if os.path.exists(candidate):
        return candidate
    return INPUT_ARTIFACT_INDEX.get(rel_path)


def save_json(obj, rel_path, compress=False):
    path = artifact_write_path(rel_path)
    os.makedirs(os.path.dirname(path), exist_ok=True)

    if compress:
        with gzip.open(path, "wt", encoding="utf-8") as f:
            json.dump(obj, f)
    else:
        with open(path, "w", encoding="utf-8") as f:
            json.dump(obj, f)

    return path


def load_json(rel_path, compress=False):
    path = artifact_read_path(rel_path)
    if path is None:
        raise FileNotFoundError(f"Artifact not found: {rel_path}")

    if compress:
        with gzip.open(path, "rt", encoding="utf-8") as f:
            return json.load(f)
    else:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)


def save_games_table_csv(games_data, name, n_games, base_seed, max_steps):
    """
    Save a user-friendly per-game CSV table into Output.
    """
    rel_path = games_table_rel_path(name, n_games, base_seed, max_steps)
    path = artifact_write_path(rel_path)
    os.makedirs(os.path.dirname(path), exist_ok=True)

    df = pd.DataFrame([
        {
            "game_index": g["game_index"],
            "seed": g["seed"],
            "final_score": g["final_score"],
            "max_tile": g["max_tile"],
            "move_count": g["move_count"],
            "duration_sec": g["duration_sec"],
            "won": g["won"],
        }
        for g in games_data
    ])

    df.to_csv(path, index=False)
    return path


def _normalize_loaded_summary(summary):
    """
    JSON turns dict keys into strings. Convert tile_counts keys back to int.
    """
    if "tile_counts" in summary and isinstance(summary["tile_counts"], dict):
        summary["tile_counts"] = {int(k): v for k, v in summary["tile_counts"].items()}
    return summary

def summarize_games(games):
    """
    Compute summary statistics from per-game results.
    """
    if not games:
        raise ValueError("Cannot summarize empty game list.")

    scores = [g["final_score"] for g in games]
    max_tiles = [g["max_tile"] for g in games]
    move_counts = [g["move_count"] for g in games]
    durations = [g["duration_sec"] for g in games]
    wins = sum(g["won"] for g in games)

    best_max_tile = max(max_tiles)
    best_tile_count = sum(1 for x in max_tiles if x == best_max_tile)

    return {
        "games": len(games),
        "avg_score": sum(scores) / len(scores),
        "best_score": max(scores),
        "avg_max_tile": sum(max_tiles) / len(max_tiles),
        "best_max_tile": best_max_tile,
        "best_tile_count": best_tile_count,
        "avg_move_count": sum(move_counts) / len(move_counts),
        "avg_duration_sec": sum(durations) / len(durations),
        "wins": wins,
        "tile_counts": dict(Counter(max_tiles)),
    }


def strip_frames_from_replay(replay):
    """
    Keep only summary-level info for games.json.gz.
    """
    return {
        "game_index": replay["game_index"],
        "seed": replay["seed"],
        "final_score": replay["final_score"],
        "max_tile": replay["max_tile"],
        "move_count": replay["move_count"],
        "won": replay["won"],
        "duration_sec": replay["duration_sec"],
    }


def reached_at_least_df(artifact, thresholds=(128, 256, 512, 1024, 2048, 4096, 8192, 16384)):
    games = artifact["games_data"]
    total = len(games)

    rows = []
    for threshold in thresholds:
        count = sum(1 for g in games if g["max_tile"] >= threshold)
        rows.append({
            "threshold": threshold,
            "count": count,
            "percent": 100.0 * count / total,
        })

    return pd.DataFrame(rows)

def render_board_html_from_bitboard(bitboard):
    board = bitboard_to_board(int(bitboard))
    cells = []

    for row in board:
        for value in row:
            color = tile_colors.get(value, "#111111")
            text_color = "#776e65" if value <= 4 else "white"
            text = "" if value == 0 else str(value)

            cells.append(
                f"""
                <div style="
                    width:68px;
                    height:68px;
                    border-radius:6px;
                    background:{color};
                    color:{text_color};
                    display:flex;
                    align-items:center;
                    justify-content:center;
                    font-weight:700;
                    font-size:20px;
                    font-family:Arial,sans-serif;
                ">{text}</div>
                """
            )

    return f"""
    <div style="
        display:grid;
        grid-template-columns:repeat(4, 68px);
        gap:6px;
        background:#bbada0;
        padding:8px;
        border-radius:8px;
        width:max-content;
    ">
        {''.join(cells)}
    </div>
    """

def play_one_game_bitboard(
    agent_fn,
    seed=None,
    game_index=None,
    max_steps=MAX_STEPS,
    collect_frames=False,
):
    """
    Play one full game using a bitboard-based agent.
    The agent must have signature: agent_fn(bitboard, legal_moves, rng)
    """
    rng = random.Random(seed)
    env = Game2048Env(rng=rng)

    # Keep this reset to preserve the old seed-to-game mapping used by your notebook.
    env.reset()

    max_tile_fn = globals().get("get_max_tile_fast", get_max_tile)

    start_time = time.perf_counter()

    frames = None
    if collect_frames:
        frames = [{
            "step": 0,
            "action": None,
            "reward": 0,
            "score": env.get_score(),
            "max_tile": max_tile_fn(env.board),
            "bitboard": env.board,
        }]

    while not env.done and env.get_move_count() < max_steps:
        bitboard = env.board
        legal_moves = env.get_legal_moves()

        action = agent_fn(bitboard, legal_moves, rng)
        if action not in legal_moves:
            raise ValueError(
                f"Agent returned illegal move '{action}' for legal moves {legal_moves}"
            )

        _, reward, done, info = env.step_bitboard(action, include_legal_moves=False)

        if collect_frames:
            frames.append({
                "step": info["move_count"],
                "action": action,
                "reward": reward,
                "score": info["score"],
                "max_tile": info["max_tile"],
                "bitboard": env.board,
            })

    duration_sec = time.perf_counter() - start_time

    result = {
        "game_index": game_index,
        "seed": seed,
        "final_score": env.get_score(),
        "max_tile": max_tile_fn(env.board),
        "move_count": env.get_move_count(),
        "won": env.won,
        "duration_sec": duration_sec,
    }

    if collect_frames:
        result["frames"] = frames

    return result

# Experiment run/load helpers
# -------------------------------------------------------------------

def run_experiment(
    name,
    agent_fn,
    n_games=100,
    base_seed=BASE_SEED,
    max_steps=MAX_STEPS,
    save_replays=False,
    force=False,
):
    """
    Run or load an experiment.

    Always saves:
    - summary.json
    - games.json.gz
    - games_table.csv

    Saves replays.json.gz only if save_replays=True.
    """
    cache_key = artifact_cache_key(name, n_games, base_seed, max_steps)
    paths = artifact_rel_paths(name, n_games, base_seed, max_steps)

    if cache_key in EXPERIMENT_MEMORY_CACHE and not force:
        cached = EXPERIMENT_MEMORY_CACHE[cache_key]
        if save_replays and "replays" not in cached:
            pass
        else:
            print(f"[MEMORY] {name} loaded from in-session cache")
            return cached

    summary_exists = artifact_read_path(paths["summary"]) is not None
    games_exists = artifact_read_path(paths["games"]) is not None
    replays_exists = artifact_read_path(paths["replays"]) is not None

    can_load_from_artifacts = (
        summary_exists and games_exists and (replays_exists or not save_replays)
    )

    if can_load_from_artifacts and not force:
        print(f"[LOAD] {name} from artifact cache")

        artifact = {
            "name": normalize_experiment_name(name),
            "n_games": n_games,
            "base_seed": base_seed,
            "max_steps": max_steps,
            "summary": _normalize_loaded_summary(load_json(paths["summary"], compress=False)),
            "games_data": load_json(paths["games"], compress=True),
            "games_table_csv": games_table_rel_path(name, n_games, base_seed, max_steps),
        }

        if save_replays and replays_exists:
            artifact["replays"] = load_json(paths["replays"], compress=True)

        EXPERIMENT_MEMORY_CACHE[cache_key] = artifact
        return artifact

    seeds = [base_seed + i for i in range(n_games)]

    results = []
    total_time = 0.0
    completed_games = 0

    parallel = Parallel(
        n_jobs=N_JOBS,
        prefer="processes",
        return_as="generator_unordered",
    )

    with tqdm(total=n_games, desc=f"Evaluating {name}") as pbar:
        for result in parallel(
            delayed(play_one_game_bitboard)(
                agent_fn=agent_fn,
                seed=seed,
                game_index=i + 1,
                max_steps=max_steps,
                collect_frames=save_replays,
            )
            for i, seed in enumerate(seeds)
        ):
            results.append(result)

            completed_games += 1
            total_time += result["duration_sec"]
            avg_game_sec = total_time / completed_games

            pbar.update(1)
            pbar.set_postfix(avg_game_s=f"{avg_game_sec:.2f}")

    results.sort(key=lambda x: x["game_index"])

    if save_replays:
        replays = results
        games_data = [strip_frames_from_replay(x) for x in replays]
    else:
        replays = None
        games_data = results

    summary = summarize_games(games_data)

    save_json(summary, paths["summary"], compress=False)
    save_json(games_data, paths["games"], compress=True)
    save_games_table_csv(games_data, name, n_games, base_seed, max_steps)

    if save_replays:
        save_json(replays, paths["replays"], compress=True)

    artifact = {
        "name": normalize_experiment_name(name),
        "n_games": n_games,
        "base_seed": base_seed,
        "max_steps": max_steps,
        "summary": summary,
        "games_data": games_data,
        "games_table_csv": games_table_rel_path(name, n_games, base_seed, max_steps),
    }

    if save_replays:
        artifact["replays"] = replays

    EXPERIMENT_MEMORY_CACHE[cache_key] = artifact
    return artifact


def load_experiment(
    name,
    n_games=100,
    base_seed=BASE_SEED,
    max_steps=MAX_STEPS,
    load_replays=True,
):
    """
    Load a previously saved experiment from memory or artifacts.
    """
    cache_key = artifact_cache_key(name, n_games, base_seed, max_steps)

    if cache_key in EXPERIMENT_MEMORY_CACHE:
        artifact = EXPERIMENT_MEMORY_CACHE[cache_key]
        if load_replays:
            return artifact
        return {k: v for k, v in artifact.items() if k != "replays"}

    paths = artifact_rel_paths(name, n_games, base_seed, max_steps)

    artifact = {
        "name": normalize_experiment_name(name),
        "n_games": n_games,
        "base_seed": base_seed,
        "max_steps": max_steps,
        "summary": _normalize_loaded_summary(load_json(paths["summary"], compress=False)),
        "games_data": load_json(paths["games"], compress=True),
        "games_table_csv": games_table_rel_path(name, n_games, base_seed, max_steps),
    }

    if load_replays and artifact_read_path(paths["replays"]) is not None:
        artifact["replays"] = load_json(paths["replays"], compress=True)

    EXPERIMENT_MEMORY_CACHE[cache_key] = artifact
    return artifact

def select_replay_record(replays, target_tile=None, mode="best", rank=1):
    if not replays:
        return None

    if target_tile is None:
        target_tile = max(r["max_tile"] for r in replays)

    candidates = [r for r in replays if r["max_tile"] == target_tile]
    if not candidates:
        return None

    if mode not in {"best", "worst"}:
        raise ValueError("mode must be 'best' or 'worst'.")

    reverse = (mode == "best")
    candidates = sorted(
        candidates,
        key=lambda r: (r["final_score"], r["move_count"], r["seed"]),
        reverse=reverse,
    )

    if rank < 1 or rank > len(candidates):
        raise ValueError(f"rank must be between 1 and {len(candidates)} for target_tile={target_tile}")

    return candidates[rank - 1]


def replay_candidates_df(replays, target_tile=None, mode="best"):
    if not replays:
        return pd.DataFrame()

    if target_tile is None:
        target_tile = max(r["max_tile"] for r in replays)

    candidates = [r for r in replays if r["max_tile"] == target_tile]
    reverse = (mode == "best")

    candidates = sorted(
        candidates,
        key=lambda r: (r["final_score"], r["move_count"], r["seed"]),
        reverse=reverse,
    )

    rows = []
    for i, r in enumerate(candidates, start=1):
        rows.append({
            "rank": i,
            "game_index": r["game_index"],
            "seed": r["seed"],
            "final_score": r["final_score"],
            "max_tile": r["max_tile"],
            "move_count": r["move_count"],
            "duration_sec": r["duration_sec"],
            "won": r["won"],
        })

    return pd.DataFrame(rows)

def show_replay_buttons(replay):
    frames = replay["frames"]
    current_idx = 0

    header = widgets.HTML()
    board_widget = widgets.HTML()

    def clamp_index(idx):
        return max(0, min(len(frames) - 1, idx))

    def render_frame(idx):
        nonlocal current_idx
        current_idx = clamp_index(idx)

        frame = frames[current_idx]
        board_html = render_board_html_from_bitboard(frame["bitboard"])

        header.value = (
            f"<b>Seed:</b> {replay['seed']} | "
            f"<b>Frame:</b> {current_idx + 1}/{len(frames)} | "
            f"<b>Step:</b> {frame['step']} | "
            f"<b>Action:</b> {frame['action']} | "
            f"<b>Reward:</b> {frame['reward']} | "
            f"<b>Score:</b> {frame['score']} | "
            f"<b>Max tile:</b> {frame['max_tile']}"
        )

        board_widget.value = board_html

    def make_button(text, delta):
        button = widgets.Button(description=text)
        button.on_click(lambda _: render_frame(current_idx + delta))
        return button

    btn_first = widgets.Button(description="First")
    btn_last = widgets.Button(description="Last")
    btn_m100 = make_button("-100", -100)
    btn_m10 = make_button("-10", -10)
    btn_prev = make_button("Prev", -1)
    btn_next = make_button("Next", 1)
    btn_p10 = make_button("+10", 10)
    btn_p100 = make_button("+100", 100)

    btn_first.on_click(lambda _: render_frame(0))
    btn_last.on_click(lambda _: render_frame(len(frames) - 1))

    controls = widgets.HBox([btn_first, btn_m100, btn_m10, btn_prev, btn_next, btn_p10, btn_p100, btn_last])

    display(header)
    display(controls)
    display(board_widget)

    render_frame(0)


def show_saved_replay(
    name,
    n_games=100,
    base_seed=BASE_SEED,
    max_steps=MAX_STEPS,
    target_tile=None,
    mode="best",
    rank=1,
    show_table=False,
):
    artifact = load_experiment(
        name=name,
        n_games=n_games,
        base_seed=base_seed,
        max_steps=max_steps,
        load_replays=True,
    )

    if "replays" not in artifact:
        print("No saved replays found for this experiment. Re-run it with save_replays=True.")
        return None

    replays = artifact["replays"]

    if show_table:
        display(replay_candidates_df(replays, target_tile=target_tile, mode=mode))

    replay = select_replay_record(
        replays=replays,
        target_tile=target_tile,
        mode=mode,
        rank=rank,
    )

    if replay is None:
        print("No matching replay found.")
        return None

    print({
        "selected_game_index": replay["game_index"],
        "selected_seed": replay["seed"],
        "selected_score": replay["final_score"],
        "selected_max_tile": replay["max_tile"],
        "selected_move_count": replay["move_count"],
        "selected_duration_sec": replay["duration_sec"],
        "mode": mode,
        "rank": rank,
    })

    show_replay_buttons(replay)
    return replay


def rebuild_replay(
    name,
    agent_fn,
    n_games=100,
    base_seed=BASE_SEED,
    max_steps=MAX_STEPS,
    target_tile=None,
    mode="best",
    rank=1,
    show_table=False,
    save_replay=True,
    return_replay=False,
):
    """
    Rebuild one replay from a saved summary-only experiment.
    """
    artifact = load_experiment(
        name=name,
        n_games=n_games,
        base_seed=base_seed,
        max_steps=max_steps,
        load_replays=False,
    )

    games_data = artifact["games_data"]
    if not games_data:
        print("No saved games_data found.")
        return None

    if target_tile is None:
        target_tile = max(g["max_tile"] for g in games_data)

    candidates = [g for g in games_data if g["max_tile"] == target_tile]
    if not candidates:
        print(f"No games found with max_tile == {target_tile}.")
        return None

    if mode not in {"best", "worst"}:
        raise ValueError("mode must be 'best' or 'worst'.")

    reverse = (mode == "best")
    candidates = sorted(
        candidates,
        key=lambda g: (g["final_score"], g["move_count"], g["seed"]),
        reverse=reverse,
    )

    if rank < 1 or rank > len(candidates):
        raise ValueError(f"rank must be between 1 and {len(candidates)}.")

    if show_table:
        display(pd.DataFrame([
            {
                "rank": i + 1,
                "game_index": g["game_index"],
                "seed": g["seed"],
                "final_score": g["final_score"],
                "max_tile": g["max_tile"],
                "move_count": g["move_count"],
                "duration_sec": g["duration_sec"],
                "won": g["won"],
            }
            for i, g in enumerate(candidates)
        ]))

    selected = candidates[rank - 1]

    replay = play_one_game_bitboard(
        agent_fn=agent_fn,
        seed=selected["seed"],
        game_index=selected["game_index"],
        max_steps=max_steps,
        collect_frames=True,
    )

    if save_replay:
        paths = artifact_rel_paths(name, n_games, base_seed, max_steps)
        existing = None

        if artifact_read_path(paths["replays"]) is not None:
            existing = load_json(paths["replays"], compress=True)

        if existing is None:
            existing = []

        existing = [r for r in existing if r["seed"] != replay["seed"]]
        existing.append(replay)
        existing.sort(key=lambda x: x["game_index"])

        save_json(existing, paths["replays"], compress=True)

    print({
        "selected_game_index": selected["game_index"],
        "selected_seed": selected["seed"],
        "selected_score": selected["final_score"],
        "selected_max_tile": selected["max_tile"],
        "selected_move_count": selected["move_count"],
        "mode": mode,
        "rank": rank,
    })

    show_replay_buttons(replay)

    if return_replay:
        return replay

    return None


def compare_models_simple(comparison_specs, max_steps=MAX_STEPS):
    """
    comparison_specs: list of (name, n_games)
    Example:
        [("random", 100), ("greedy", 100)]
    """
    rows = []

    for name, n_games in comparison_specs:
        artifact = load_experiment(
            name=name,
            n_games=n_games,
            base_seed=BASE_SEED,
            max_steps=max_steps,
            load_replays=False,
        )

        s = artifact["summary"]

        rows.append({
            "agent": f"{name}_{n_games}",
            "best_score": s["best_score"],
            "avg_score": round(s["avg_score"], 2),
            "best_max_tile": s["best_max_tile"],
            "avg_max_tile": round(s["avg_max_tile"], 2),
            "avg_move_count": round(s["avg_move_count"], 2),
            "wins": s["wins"],
        })

    comparison_df = pd.DataFrame(rows)
    display(comparison_df)
    return comparison_df
