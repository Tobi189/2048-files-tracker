# This cell defines lookahead version 7 and caches:
# - 100-game results with full replays (if desired)
# - 500-game results as the main strength evaluation
# By default, only 500-game summaries are saved for the heavier run.

def evaluate_sequence_v7(board, depth):
    """
    Depth-limited recursive lookahead built on heuristic v5.
    """
    if depth == 0:
        return heuristic_score_v5(board)

    legal_moves = get_legal_moves(board)
    if not legal_moves:
        return heuristic_score_v5(board)

    best_value = float("-inf")

    for move in legal_moves:
        next_board, reward, changed = apply_move(board, move)
        if not changed:
            continue

        value = 3.0 * reward + evaluate_sequence_v7(next_board, depth - 1)

        if value > best_value:
            best_value = value

    if best_value == float("-inf"):
        return heuristic_score_v5(board)

    return best_value

def lookahead_agent_v7(board, legal_moves, rng=None):
    """
    Three-ply lookahead agent built on heuristic v5.
    """
    rng = rng or random
    best_moves = []
    best_value = float("-inf")

    for move in legal_moves:
        next_board, reward, changed = apply_move(board, move)
        if not changed:
            continue

        value = heuristic_score_v5(next_board) + 3.0 * reward + 0.8 * evaluate_sequence_v7(next_board, depth=2)

        if value > best_value:
            best_value = value
            best_moves = [move]
        elif value == best_value:
            best_moves.append(move)

    return rng.choice(best_moves) if best_moves else rng.choice(legal_moves)

LOOKAHEAD_V7_GAMES_100 = 100
LOOKAHEAD_V7_GAMES_500 = 500

lookahead_v7_artifact_100 = run_experiment_cached(
    name="lookahead_v7",
    agent_fn=lookahead_agent_v7,
    n_games=LOOKAHEAD_V7_GAMES_100,
    base_seed=BASE_SEED,
    save_replays=SAVE_REPLAYS_FOR_100,
    force=False,
)

lookahead_v7_artifact_500 = run_experiment_cached(

    n_games=LOOKAHEAD_V7_GAMES_500,
