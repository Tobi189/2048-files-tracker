def log2_tile_lookahead_v1(value):
    if value <= 0:
        return 0
    return value.bit_length() - 1


def snake_path_score_lookahead_v1(bitboard):
    board = bitboard_to_board(bitboard)
    path = [
        (3, 3), (3, 2), (3, 1), (3, 0),
        (2, 0), (2, 1), (2, 2), (2, 3),
        (1, 3), (1, 2), (1, 1), (1, 0),
        (0, 0), (0, 1), (0, 2), (0, 3),
    ]
    vals = [log2_tile_lookahead_v1(board[r][c]) for r, c in path]

    score = 0
    for i in range(len(vals) - 1):
        if vals[i] >= vals[i + 1]:
            score += vals[i] - vals[i + 1] + 1
        else:
            score -= 4 * (vals[i + 1] - vals[i])

    return score


def merge_potential_score_lookahead_v1(bitboard):
    board = bitboard_to_board(bitboard)
    score = 0

    for r in range(4):
        for c in range(3):
            a, b = board[r][c], board[r][c + 1]
            if a != 0 and b != 0:
                la, lb = log2_tile_lookahead_v1(a), log2_tile_lookahead_v1(b)
                if a == b:
                    score += 12
                elif abs(la - lb) == 1:
                    score += 3

    for c in range(4):
        for r in range(3):
            a, b = board[r][c], board[r + 1][c]
            if a != 0 and b != 0:
                la, lb = log2_tile_lookahead_v1(a), log2_tile_lookahead_v1(b)
                if a == b:
                    score += 10
                elif abs(la - lb) == 1:
                    score += 2

    return score


def bottom_two_rows_score_lookahead_v1(bitboard):
    board = bitboard_to_board(bitboard)
    score = 0

    for r in range(4):
        for c in range(4):
            val = log2_tile_lookahead_v1(board[r][c])
            if r >= 2:
                score += 2 * val
            else:
                score -= 1 * val

    return score


def emptiness_balance_score_lookahead_v1(bitboard):
    board = bitboard_to_board(bitboard)
    empty_cells = 0
    row_fills = []

    for row in board:
        filled = 0
        for x in row:
            if x == 0:
                empty_cells += 1
            else:
                filled += 1
        row_fills.append(filled)

    imbalance = max(row_fills) - min(row_fills)
    return 20 * empty_cells - 5 * imbalance


def corner_lock_score_lookahead_v1(bitboard):
    board = bitboard_to_board(bitboard)
    max_tile = get_max_tile(bitboard)
    score = 0

    if board[3][3] == max_tile:
        score += 80

    score += 8 * log2_tile_lookahead_v1(board[3][2])
    score += 8 * log2_tile_lookahead_v1(board[2][3])
    score += 4 * log2_tile_lookahead_v1(board[3][1])
    score += 4 * log2_tile_lookahead_v1(board[1][3])

    return score


def heuristic_score_lookahead_v1(bitboard):
    max_tile = get_max_tile(bitboard)

    return (
        120 * log2_tile_lookahead_v1(max_tile)
        + 60 * snake_path_score_lookahead_v1(bitboard)
        + 35 * merge_potential_score_lookahead_v1(bitboard)
        + 25 * bottom_two_rows_score_lookahead_v1(bitboard)
        + 15 * emptiness_balance_score_lookahead_v1(bitboard)
        + 40 * corner_lock_score_lookahead_v1(bitboard)
    )


def two_step_value_lookahead_v1(bitboard):
    legal_1 = get_legal_moves(bitboard)
    if not legal_1:
        return heuristic_score_lookahead_v1(bitboard)

    best_second = float("-inf")

    for move2 in legal_1:
        board2, reward2, changed2 = apply_move_bitboard(bitboard, move2)
        if not changed2:
            continue

        val2 = heuristic_score_lookahead_v1(board2) + 2.0 * reward2
        if val2 > best_second:
            best_second = val2

    if best_second == float("-inf"):
        best_second = heuristic_score_lookahead_v1(bitboard)

    return heuristic_score_lookahead_v1(bitboard) + 0.6 * best_second


def lookahead_agent_v1(bitboard, legal_moves, rng=None):
    """
    Choose the move with the best two-step evaluation.
    """
    rng = rng or random
    best_moves = []
    best_value = float("-inf")

    for move in legal_moves:
        board1, reward1, changed1 = apply_move_bitboard(bitboard, move)
        if not changed1:
            continue

        value = two_step_value_lookahead_v1(board1) + 3.0 * reward1

        if value > best_value:
            best_value = value
            best_moves = [move]
        elif value == best_value:
            best_moves.append(move)

    return rng.choice(best_moves) if best_moves else rng.choice(legal_moves)
