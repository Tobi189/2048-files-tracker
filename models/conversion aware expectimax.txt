CONVERSION_AWARE_EXPECTIMAX = "conversion_aware_expectimax"

SECM_V1_REWARD_WEIGHT = 2.2
SECM_V1_AFTERSTATE_WEIGHT = 0.18
SECM_V1_BASE_PROBABILITY_CUTOFF = 0.01
SECM_V1_MAX_DEPTH = 4

# Persistent deterministic caches.
# These do not change model logic because board info and board score are deterministic.
SECM_V1_GLOBAL_EVAL_CACHE = {}
SECM_V1_GLOBAL_INFO_CACHE = {}
SECM_V1_MAX_GLOBAL_CACHE_SIZE = 750_000

# Precomputed distance to bottom-right anchor region: cells 15, 14, 11, 10.
SECM_V1_ANCHOR_DISTANCE = []
for _idx in range(16):
    _r = _idx >> 2
    _c = _idx & 3
    SECM_V1_ANCHOR_DISTANCE.append(
        min(
            abs(_r - 3) + abs(_c - 3),
            abs(_r - 3) + abs(_c - 2),
            abs(_r - 2) + abs(_c - 3),
            abs(_r - 2) + abs(_c - 2),
        )
    )


# ------------------------------------------------------------
# Cache management
# ------------------------------------------------------------

def maybe_trim_secm_v1_global_caches():
    if len(SECM_V1_GLOBAL_EVAL_CACHE) > SECM_V1_MAX_GLOBAL_CACHE_SIZE:
        SECM_V1_GLOBAL_EVAL_CACHE.clear()

    if len(SECM_V1_GLOBAL_INFO_CACHE) > SECM_V1_MAX_GLOBAL_CACHE_SIZE:
        SECM_V1_GLOBAL_INFO_CACHE.clear()


def clear_conversion_aware_expectimax_caches():
    SECM_V1_GLOBAL_EVAL_CACHE.clear()
    SECM_V1_GLOBAL_INFO_CACHE.clear()
    print("[OK] conversion-aware expectimax global caches cleared")

def cell_log_secm_v1(bitboard, idx):
    return (bitboard >> (4 * idx)) & 0xF


def idx_row_secm_v1(idx):
    return idx >> 2


def idx_col_secm_v1(idx):
    return idx & 3


def manhattan_secm_v1(a, b):
    return abs((a >> 2) - (b >> 2)) + abs((a & 3) - (b & 3))


def anchor_region_distance_secm_v1(idx):
    return SECM_V1_ANCHOR_DISTANCE[idx]


# ------------------------------------------------------------
# Board info cache
# ------------------------------------------------------------

def board_info_secm_v1(bitboard, info_cache):
    cached = info_cache.get(bitboard)
    if cached is not None:
        return cached

    empties = 0
    mask = 0

    max1 = -1
    pos1 = -1
    max2 = -1
    pos2 = -1

    for idx in range(16):
        v = (bitboard >> (4 * idx)) & 0xF

        if v == 0:
            empties += 1
        else:
            mask |= 1 << v

        if v > max1:
            max2 = max1
            pos2 = pos1
            max1 = v
            pos1 = idx
        elif v > max2:
            max2 = v
            pos2 = idx

    unique_logs = mask.bit_count()

    if max1 < 10:
        conversion_mode = False
    elif max2 < max1 - 1:
        conversion_mode = False
    elif SECM_V1_ANCHOR_DISTANCE[pos1] <= 2 or SECM_V1_ANCHOR_DISTANCE[pos2] <= 2:
        conversion_mode = True
    elif empties <= 3:
        conversion_mode = True
    else:
        conversion_mode = False

    reward_weight = 1.6 if conversion_mode else SECM_V1_REWARD_WEIGHT

    if not conversion_mode:
        probability_cutoff = SECM_V1_BASE_PROBABILITY_CUTOFF
    else:
        if max1 >= 11 or empties <= 2:
            probability_cutoff = 0.003
        else:
            probability_cutoff = 0.006

    if not conversion_mode:
        internal_top_k = 2
    else:
        if max1 >= 11 or empties <= 2:
            internal_top_k = 4
        else:
            internal_top_k = 3

    if empties >= 6:
        depth = 2
    else:
        depth = 3

    if empties <= 2 and unique_logs <= 5:
        depth = 3

    if conversion_mode and max1 >= 11 and empties <= 3:
        depth = 4

    depth = min(depth, SECM_V1_MAX_DEPTH)

    if max1 < 10:
        conversion_bonus = 0.0
    else:
        conversion_bonus = 0.0

        if max2 == max1:
            conversion_bonus += 60.0
        elif max2 == max1 - 1:
            conversion_bonus += 28.0
        else:
            conversion_bonus -= 12.0 * (max1 - max2 - 1)

        if pos1 >= 0 and pos2 >= 0:
            d_pair = abs((pos1 >> 2) - (pos2 >> 2)) + abs((pos1 & 3) - (pos2 & 3))
            conversion_bonus += max(0.0, 12.0 - 4.0 * d_pair)

        d1 = SECM_V1_ANCHOR_DISTANCE[pos1] if pos1 >= 0 else 99
        d2 = SECM_V1_ANCHOR_DISTANCE[pos2] if pos2 >= 0 else 99

        conversion_bonus += max(0.0, 10.0 - 3.0 * d1)
        conversion_bonus += max(0.0, 8.0 - 2.0 * d2)

        corner = (bitboard >> 60) & 0xF
        left_of_corner = (bitboard >> 56) & 0xF
        above_corner = (bitboard >> 44) & 0xF
        diag = (bitboard >> 40) & 0xF

        if corner == max1 and max1 > 0:
            conversion_bonus += 24.0

        conversion_bonus += 4.0 * left_of_corner
        conversion_bonus += 4.0 * above_corner
        conversion_bonus += 1.5 * diag

        if left_of_corner > 0 and above_corner > 0:
            if left_of_corner == above_corner:
                conversion_bonus += 8.0
            elif abs(left_of_corner - above_corner) == 1:
                conversion_bonus += 3.0

    info = {
        "empties": empties,
        "unique_logs": unique_logs,
        "max1": max1,
        "pos1": pos1,
        "max2": max2,
        "pos2": pos2,
        "conversion_mode": conversion_mode,
        "reward_weight": reward_weight,
        "probability_cutoff": probability_cutoff,
        "internal_top_k": internal_top_k,
        "depth": depth,
        "conversion_bonus": conversion_bonus,
    }

    info_cache[bitboard] = info
    return info

# Fast board scoring
# ------------------------------------------------------------

def evaluate_board_fast_secm_v1(bitboard, info):
    """
    Logic-equivalent to:
        evaluate_board_fast(bitboard)

    but avoids a second max-tile scan inside corner_anchor_bonus by using info["max1"].
    """
    score = 0.0

    for row_idx in range(4):
        row16 = (bitboard >> (16 * row_idx)) & ROW_MASK_16
        score += ROW_TOTAL_SCORE[row16]

    transposed = _transpose_bitboard_fast(bitboard)
    for row_idx in range(4):
        row16 = (transposed >> (16 * row_idx)) & ROW_MASK_16
        score += ROW_TOTAL_SCORE[row16]

    max_log = info["max1"]
    corner = (bitboard >> 60) & 0xF

    if corner == max_log and max_log > 0:
        score += 40.0

    score += 4.0 * ((bitboard >> 56) & 0xF)
    score += 4.0 * ((bitboard >> 44) & 0xF)

    return score


def score_board_secm_v1_cached(bitboard, eval_cache, info_cache):
    cached = eval_cache.get(bitboard)
    if cached is not None:
        return cached

    info = board_info_secm_v1(bitboard, info_cache)
    value = evaluate_board_fast_secm_v1(bitboard, info) + info["conversion_bonus"]

    eval_cache[bitboard] = value
    return value

def move_bias_secm_v1_cached(bitboard, move, info_cache):
    conversion_mode = board_info_secm_v1(bitboard, info_cache)["conversion_mode"]

    if conversion_mode:
        if move == "down":
            return 1.5
        if move == "right":
            return 1.5
        if move == "left":
            return 0.0
        return -1.0

    if move == "down":
        return 2.0
    if move == "right":
        return 2.0
    if move == "left":
        return 0.0
    return -3.0


def move_bias_from_info_secm_v1(info, move):

        if move == "down":
            return 1.5

            return 1.5
        if move == "left":
            return 0.0
        return -1.0

    if move == "down":
        return 2.0
    if move == "right":
        return 2.0
    if move == "left":
        return 0.0
    return -3.0




# ------------------------------------------------------------
# Candidate generation and hot-path move helpers
# ------------------------------------------------------------

SECM_V1_MOVE_ORDER = ("down", "right", "left", "up")


def _apply_ordered_move_secm_v1(bitboard, move):
    """
    Same logic as apply_move_bitboard, but avoids string normalization/dispatch overhead
    because move is already one of the known internal move strings.
    """
    if move == "down":
        return move_down_bitboard_fast(bitboard)
    if move == "right":
        return move_right_bitboard_fast(bitboard)
    if move == "left":
        return move_left_bitboard_fast(bitboard)
    return move_up_bitboard_fast(bitboard)


def _move_bias_from_conversion_secm_v1(conversion_mode, move):
    if conversion_mode:
        if move == "down":
            return 1.5
        if move == "right":
            return 1.5
        if move == "left":
            return 0.0
        return -1.0

    if move == "down":
        return 2.0
    if move == "right":
        return 2.0
    if move == "left":
        return 0.0
    return -3.0


def _finalize_candidates_secm_v1(c1, c2, c3, c4, top_k):
    """
    Avoid generator/list overhead for a max of four candidates.
    """
    if c1 is None:
        return ()
    if c2 is None:
        return (c1,)
    if c3 is None:
        return (c1, c2)[:top_k]
    if c4 is None:
        return (c1, c2, c3)[:top_k]
    return (c1, c2, c3, c4)[:top_k]


def generate_candidates_scored_secm_v1(bitboard, legal_moves, eval_cache, info_cache, top_k):
    """
    Same candidate logic as before, but each candidate also stores after_score
    so recursive search does not need another score_board cache lookup.
    
    Candidate format:
        (quick_score, move, next_bitboard, reward, after_score)
    """
    c1 = None
    c2 = None
    c3 = None
    c4 = None

    info = board_info_secm_v1(bitboard, info_cache)
    rw = info["reward_weight"]
    conversion_mode = info["conversion_mode"]

    for move in SECM_V1_MOVE_ORDER:
        if legal_moves is not None and move not in legal_moves:
            continue

        next_bitboard, reward, changed = _apply_ordered_move_secm_v1(bitboard, move)
        if not changed:
            continue

        after_score = score_board_secm_v1_cached(next_bitboard, eval_cache, info_cache)
        quick_score = (
            after_score
            + rw * reward
            + _move_bias_from_conversion_secm_v1(conversion_mode, move)
        )

        cand = (quick_score, move, next_bitboard, reward, after_score)
        c1, c2, c3, c4 = _insert_candidate_desc_stable_secm_v1(c1, c2, c3, c4, cand)

    return _finalize_candidates_secm_v1(c1, c2, c3, c4, top_k)


def generate_candidates_secm_v1(bitboard, legal_moves, eval_cache, info_cache, top_k):
    """
    Compatibility wrapper preserving the old 4-field candidate format.
    """
    scored = generate_candidates_scored_secm_v1(
        bitboard,
        legal_moves,
        eval_cache,
        info_cache,
        top_k,
    )

    return tuple(
        (quick_score, move, next_bitboard, reward)
        for quick_score, move, next_bitboard, reward, after_score in scored
    )


def player_value_secm_v1_exact_fast(bitboard, depth, cumulative_prob, tt_player, tt_chance, eval_cache, info_cache):
    key = (bitboard, depth)
    cached = tt_player.get(key)
    if cached is not None:
        return cached

    if depth == 0:
        value = score_board_secm_v1_cached(bitboard, eval_cache, info_cache)
        tt_player[key] = value
        return value

    info = board_info_secm_v1(bitboard, info_cache)
    top_k = info["internal_top_k"]
    rw = info["reward_weight"]

    # Logic-equivalent to get_legal_moves + generate_candidates:
    # if no move changes the board, candidates will be empty.
    candidates = generate_candidates_scored_secm_v1(
        bitboard,
        legal_moves=None,
        eval_cache=eval_cache,
        info_cache=info_cache,
        top_k=top_k,
    )

    if not candidates:
        value = score_board_secm_v1_cached(bitboard, eval_cache, info_cache)
        tt_player[key] = value
        return value

    best_value = float("-inf")

    for _, move, next_bitboard, reward, after_score in candidates:
        value = (
            rw * reward
            + SECM_V1_AFTERSTATE_WEIGHT * after_score
            + chance_value_secm_v1_exact_fast(
                next_bitboard,
                depth,
                cumulative_prob,
                tt_player,
                tt_chance,
                eval_cache,
                info_cache,
            )
        )

        if value > best_value:
            best_value = value

    tt_player[key] = best_value
    return best_value




# ------------------------------------------------------------
# Chance node
# ------------------------------------------------------------

def chance_value_secm_v1_exact_fast(afterstate_bitboard, depth, cumulative_prob, tt_player, tt_chance, eval_cache, info_cache):
    info = board_info_secm_v1(afterstate_bitboard, info_cache)
    cutoff = info["probability_cutoff"]

    if depth <= 0 or cumulative_prob < cutoff:
        return score_board_secm_v1_cached(afterstate_bitboard, eval_cache, info_cache)

    if info["conversion_mode"]:
        key = (afterstate_bitboard, depth, round(cumulative_prob, 6))
    else:
        key = (afterstate_bitboard, depth)

    cached = tt_chance.get(key)
    if cached is not None:
        return cached

    empties = info["empties"]

    if empties == 0:
        value = player_value_secm_v1_exact_fast(
            afterstate_bitboard,
            depth - 1,
            cumulative_prob,
            tt_player,
            tt_chance,
            eval_cache,
            info_cache,
        )
        tt_chance[key] = value
        return value

    p2 = 0.9 / empties
    p4 = 0.1 / empties
    total = 0.0

    for idx in range(16):
        shift = 4 * idx
        if ((afterstate_bitboard >> shift) & 0xF) != 0:
            continue

        spawned2 = afterstate_bitboard | (1 << shift)
        prob2 = cumulative_prob * p2

        if prob2 < cutoff:
            total += p2 * score_board_secm_v1_cached(spawned2, eval_cache, info_cache)
        else:
            total += p2 * player_value_secm_v1_exact_fast(
                spawned2,
                depth - 1,
                prob2,
                tt_player,
                tt_chance,
                eval_cache,
                info_cache,
            )

        spawned4 = afterstate_bitboard | (2 << shift)
        prob4 = cumulative_prob * p4

        if prob4 < cutoff:
            total += p4 * score_board_secm_v1_cached(spawned4, eval_cache, info_cache)
        else:
            total += p4 * player_value_secm_v1_exact_fast(
                spawned4,
                depth - 1,
                prob4,
                tt_player,
                tt_chance,
                eval_cache,
                info_cache,
            )

    tt_chance[key] = total
    return total




# ------------------------------------------------------------
# Agent
# ------------------------------------------------------------

def conversion_aware_expectimax(bitboard, legal_moves, rng=None):
    rng = rng or random

    maybe_trim_secm_v1_global_caches()

    tt_player = {}
    tt_chance = {}

    eval_cache = SECM_V1_GLOBAL_EVAL_CACHE
    info_cache = SECM_V1_GLOBAL_INFO_CACHE

    info = board_info_secm_v1(bitboard, info_cache)
    depth = info["depth"]
    rw = info["reward_weight"]

    candidates = generate_candidates_scored_secm_v1(
        bitboard,
        legal_moves=legal_moves,
        eval_cache=eval_cache,
        info_cache=info_cache,
        top_k=4,
    )

    best_value = float("-inf")
    best_moves = []

    for _, move, next_bitboard, reward, after_score in candidates:
        value = (
            rw * reward
            + SECM_V1_AFTERSTATE_WEIGHT * after_score
            + chance_value_secm_v1_exact_fast(
                next_bitboard,
                depth,
                1.0,
                tt_player,
                tt_chance,
                eval_cache,
                info_cache,
            )
        )

        if value > best_value:
            best_value = value
            best_moves = [move]
        elif value == best_value:
            best_moves.append(move)

    return rng.choice(best_moves) if best_moves else rng.choice(legal_moves)
