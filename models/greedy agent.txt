# Greedy baseline agent

def greedy_agent(bitboard, legal_moves, rng=None):
    """
    Choose the move with the highest immediate merge reward.
    Use number of empty cells as a tie-breaker.
    """
    rng = rng or random

    best_moves = []
    best_reward = -1
    best_empty = -1

    for move in legal_moves:
        next_bitboard, reward, changed = apply_move_bitboard(bitboard, move)

        empty_count = 0
        for idx in range(16):
            if ((next_bitboard >> (4 * idx)) & 0xF) == 0:
                empty_count += 1

        if reward > best_reward:
            best_reward = reward
            best_empty = empty_count
            best_moves = [move]
        elif reward == best_reward:
            if empty_count > best_empty:
                best_empty = empty_count
                best_moves = [move]
            elif empty_count == best_empty:
                best_moves.append(move)

    return rng.choice(best_moves)
