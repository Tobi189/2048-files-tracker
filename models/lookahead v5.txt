# This cell defines lookahead version 5 and caches its 100-game outputs and replays.

def log2_tile_v5(value):
    """
    Convert a tile value to its log2 scale.
    """
    if value <= 0:
        return 0
    return value.bit_length() - 1

def snake_path_score_v5(board):
    """
    Reward a bottom-right snake ordering through the board.
    """
    path = [
        (3, 3), (3, 2), (3, 1), (3, 0),
        (2, 0), (2, 1), (2, 2), (2, 3),
        (1, 3), (1, 2), (1, 1), (1, 0),
        (0, 0), (0, 1), (0, 2), (0, 3),
    ]
    vals = [log2_tile_v5(board[r][c]) for r, c in path]

    score = 0
    for i in range(len(vals) - 1):
        if vals[i] >= vals[i + 1]:
            score += vals[i] - vals[i + 1] + 1
        else:
            score -= 4 * (vals[i + 1] - vals[i])

    return score

def merge_potential_score_v5(board):
    """
    Reward adjacent equal or near-equal tiles that are likely to merge soon.
    """
    score = 0

    for r in range(4):
        for c in range(3):
            a, b = board[r][c], board[r][c + 1]
            if a != 0 and b != 0:
                la, lb = log2_tile_v5(a), log2_tile_v5(b)
                if a == b:
                    score += 12
                elif abs(la - lb) == 1:
                    score += 3

    for c in range(4):
        for r in range(3):
            a, b = board[r][c], board[r + 1][c]
            if a != 0 and b != 0:
                la, lb = log2_tile_v5(a), log2_tile_v5(b)
                if a == b:
                    score += 10
                elif abs(la - lb) == 1:
                    score += 2

    return score

def bottom_two_rows_score_v5(board):
    """
    Reward mass concentrated in the bottom two rows.
    """
    score = 0

    for r in range(4):
        for c in range(4):
            val = log2_tile_v5(board[r][c])
            if r >= 2:
                score += 2 * val
            else:
                score -= 1 * val

    return score

def emptiness_balance_score_v5(board):
    """
    Reward empty cells and lightly penalize lopsided row fill.
    """
    empty_cells = len(get_empty_cells(board))
    row_fills = [sum(1 for x in row if x != 0) for row in board]
    imbalance = max(row_fills) - min(row_fills)

    return 20 * empty_cells - 5 * imbalance

def corner_lock_score_v5(board):
    """
    Reward keeping the maximum tile and its support structure near the bottom-right corner.
    """
    max_tile = get_max_tile(board)
    score = 0

    if board[3][3] == max_tile:
        score += 80

    score += 8 * log2_tile_v5(board[3][2])
    score += 8 * log2_tile_v5(board[2][3])
    score += 4 * log2_tile_v5(board[3][1])
    score += 4 * log2_tile_v5(board[1][3])

    return score

def heuristic_score_v5(board):
    """
    Main v5 board evaluation function.
    """
    max_tile = get_max_tile(board)

    return (
        120 * log2_tile_v5(max_tile) +
        60 * snake_path_score_v5(board) +
        35 * merge_potential_score_v5(board) +
        25 * bottom_two_rows_score_v5(board) +
        15 * emptiness_balance_score_v5(board) +
        40 * corner_lock_score_v5(board)
    )

def two_step_value_v5(board):
    """
    Evaluate one board plus the best immediate next move using heuristic v5.
    """
    legal_1 = get_legal_moves(board)
    if not legal_1:
        return heuristic_score_v5(board)

    best_second = float("-inf")

    for move2 in legal_1:
        board2, reward2, changed2 = apply_move(board, move2)
        if not changed2:
            continue
        val2 = heuristic_score_v5(board2) + 2.0 * reward2
        if val2 > best_second:
            best_second = val2

    if best_second == float("-inf"):
        best_second = heuristic_score_v5(board)

    return heuristic_score_v5(board) + 0.6 * best_second

def lookahead_agent_v5(board, legal_moves, rng=None):
    """
    Choose the move with the best two-step v5 evaluation.
    """
    rng = rng or random
    best_moves = []
    best_value = float("-inf")

    for move in legal_moves:
        board1, reward1, changed1 = apply_move(board, move)
        if not changed1:
            continue

        value = two_step_value_v5(board1) + 3.0 * reward1

        if value > best_value:
            best_value = value
            best_moves = [move]
        elif value == best_value:
            best_moves.append(move)

    return rng.choice(best_moves) if best_moves else rng.choice(legal_moves)
