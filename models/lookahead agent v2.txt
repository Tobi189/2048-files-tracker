def evaluate_sequence_lookahead_v2(bitboard, depth):
    """
    Depth-limited recursive lookahead built on Lookahead Agent I heuristic.
    """
    if depth == 0:
        return heuristic_score_lookahead_v1(bitboard)

    legal_moves = get_legal_moves(bitboard)
    if not legal_moves:
        return heuristic_score_lookahead_v1(bitboard)

    best_value = float("-inf")

    for move in legal_moves:
        next_bitboard, reward, changed = apply_move_bitboard(bitboard, move)
        if not changed:
            continue

        value = 3.0 * reward + evaluate_sequence_lookahead_v2(next_bitboard, depth - 1)

        if value > best_value:
            best_value = value

    if best_value == float("-inf"):
        return heuristic_score_lookahead_v1(bitboard)

    return best_value


def lookahead_agent_v2(bitboard, legal_moves, rng=None):
    """
    Three-ply lookahead agent built on Lookahead Agent I heuristic.
    """
    rng = rng or random
    best_moves = []
    best_value = float("-inf")

    for move in legal_moves:
        next_bitboard, reward, changed = apply_move_bitboard(bitboard, move)
        if not changed:
            continue

        value = (
            heuristic_score_lookahead_v1(next_bitboard)
            + 3.0 * reward
            + 0.8 * evaluate_sequence_lookahead_v2(next_bitboard, depth=2)
        )

        if value > best_value:
            best_value = value
            best_moves = [move]
        elif value == best_value:
            best_moves.append(move)

    return rng.choice(best_moves) if best_moves else rng.choice(legal_moves)


LOOKAHEAD_AGENT_V2_NAME = "lookahead_agent_v2"
