# This cell defines heuristic version 3 and caches its 100-game results.

def monotonicity_score_v3(board):
    """
    Reward boards where rows and columns are more monotonic.
    """
    score = 0

    for row in board:
        for i in range(len(row) - 1):
            if row[i] >= row[i + 1]:
                score += 1

    for col in zip(*board):
        for i in range(len(col) - 1):
            if col[i] >= col[i + 1]:
                score += 1

    return score

def smoothness_score_v3(board):
    """
    Penalize large differences between neighboring tiles.
    """
    score = 0

    for r in range(4):
        for c in range(3):
            score -= abs(board[r][c] - board[r][c + 1])

    for c in range(4):
        for r in range(3):
            score -= abs(board[r][c] - board[r + 1][c])

    return score

def heuristic_score_v3(board):
    """
    Heuristic v3 combining:
    - empty cells
    - max tile
    - bottom-right corner preference
    - monotonicity
    - smoothness
    """
    empty_cells = len(get_empty_cells(board))
    max_tile = get_max_tile(board)
    corner_bonus = 1 if board[3][3] == max_tile else 0

    return (
        50 * empty_cells +
        1.0 * max_tile +
        1000 * corner_bonus +
        20 * monotonicity_score_v3(board) +
        0.1 * smoothness_score_v3(board)
    )

def heuristic_agent_v3(board, legal_moves, rng=None):
    """
    Choose the move whose resulting board has the best v3 heuristic score.
    """
    rng = rng or random
    best_moves = []
    best_value = float("-inf")

    for move in legal_moves:
        next_board, reward, changed = apply_move(board, move)
        value = heuristic_score_v3(next_board) + reward

        if value > best_value:
            best_value = value
            best_moves = [move]
        elif value == best_value:
            best_moves.append(move)

    return rng.choice(best_moves)
